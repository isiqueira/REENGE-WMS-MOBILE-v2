Create `src/app/core/services/barcode-focus.service.ts`:
```typescript
import { Injectable, OnDestroy } from '@angular/core';
import { Keyboard } from '@capacitor/keyboard';
import { Preferences } from '@capacitor/preferences';

@Injectable({ providedIn: 'root' })
export class BarcodeFocusService implements OnDestroy {
  private readonly INTERVAL_MS = 20;
  private readonly HIDE_KEYBOARD_KEY = 'hideKeyboard-sessionWms';
  private intervalId: ReturnType<typeof setInterval> | null = null;

  startFocusLoop(): void {
    if (this.intervalId !== null) return; // already running
    this.intervalId = setInterval(() => this.tick(), this.INTERVAL_MS);
  }

  stopFocusLoop(): void {
    if (this.intervalId !== null) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }

  ngOnDestroy(): void {
    this.stopFocusLoop();
  }

  private async tick(): Promise<void> {
    const focoInput = document.querySelector<HTMLElement>('input.foco, ion-input.foco');
    if (focoInput) {
      const active = document.activeElement;
      if (active !== focoInput && !(active instanceof HTMLInputElement && focoInput.contains(active))) {
        focoInput.focus();
      }
    }

    const { value } = await Preferences.get({ key: this.HIDE_KEYBOARD_KEY });
    const hideKeyboard = value !== null ? JSON.parse(value) as boolean : false;
    if (hideKeyboard) {
      await Keyboard.hide().catch(() => { /* platform may not support it */ });
    }
  }
}
```

Notes:
- `startFocusLoop()` is idempotent — safe to call multiple times
- `Keyboard.hide()` from `@capacitor/keyboard` replaces legacy `cordova-plugin-keyboard`
- `querySelector('input.foco, ion-input.foco')` replaces legacy jQuery `$("input.foco").get(0)`
- `HIDE_KEYBOARD_KEY` matches legacy key exactly
