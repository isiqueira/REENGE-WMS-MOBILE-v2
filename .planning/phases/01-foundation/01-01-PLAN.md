```typescript
import { Injectable } from '@angular/core';
import { Preferences } from '@capacitor/preferences';

export interface WmsSession {
  UserName: string;
  [key: string]: unknown;
}

@Injectable({ providedIn: 'root' })
export class SessionService {
  private readonly SESSION_KEY = 'user-sessionWms';

  async getSession(): Promise<WmsSession | null> {
    const { value } = await Preferences.get({ key: this.SESSION_KEY });
    if (!value) return null;
    return JSON.parse(value) as WmsSession;
  }

  async setSession(session: WmsSession): Promise<void> {
    await Preferences.set({ key: this.SESSION_KEY, value: JSON.stringify(session) });
  }

  async clearSession(): Promise<void> {
    await Preferences.remove({ key: this.SESSION_KEY });
  }

  async hasSession(): Promise<boolean> {
    const { value } = await Preferences.get({ key: this.SESSION_KEY });
    return value !== null;
  }
}
```

Notes:
- Uses `@capacitor/preferences` (already in package.json as `@capacitor/preferences`)
- Session key `user-sessionWms` matches legacy exactly
- `WmsSession` uses `[key: string]: unknown` to accommodate the dynamic fields the API returns

```typescript
import { Injectable } from '@angular/core';
import { HttpClient, HttpHeaders } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';
import { Preferences } from '@capacitor/preferences';
import { SessionService } from './session.service';

@Injectable({ providedIn: 'root' })
export class ApiService {
  private readonly DEFAULT_URL = 'http://api.mayanwms.com.br/log/api';
  private readonly URL_KEY = 'apiUrl-sessionWms';

  constructor(
    private http: HttpClient,
    private sessionService: SessionService,
  ) {}

  private async getBaseUrl(): Promise<string> {
    const { value } = await Preferences.get({ key: this.URL_KEY });
    let url = value ?? this.DEFAULT_URL;
    if (!url.endsWith('/')) url += '/';
    return url;
  }

  async setBaseUrl(url: string): Promise<void> {
    await Preferences.set({ key: this.URL_KEY, value: url });
  }

  async getApi(endpoint: string, params?: Record<string, unknown>, logged = true): Promise<unknown> {
    const url = await this.getBaseUrl();
    const qs = await this.buildQueryString(params ?? {}, logged);
    return firstValueFrom(this.http.get(`${url}${endpoint}/?${qs}`));
  }

  async postApi(endpoint: string, body: Record<string, unknown> = {}, logged = true): Promise<unknown> {
    const url = await this.getBaseUrl();
    const headers = new HttpHeaders({ 'Content-Type': 'application/json' });
    if (logged) {
      const session = await this.sessionService.getSession();
      body['UserNameCreator'] = session?.UserName ?? '';
    }
    return firstValueFrom(this.http.post(`${url}${endpoint}/`, body, { headers }));
  }

  private async buildQueryString(params: Record<string, unknown>, logged: boolean): Promise<string> {
    if (logged) {
      const session = await this.sessionService.getSession();
      params['UserName'] = session?.UserName ?? '';
    }
    return Object.entries(params)
      .map(([k, v]) => `${k}=${v}`)
      .join('&');
  }
}
```

Notes:
- `firstValueFrom` replaces the Observable `.subscribe()` wrapper pattern from legacy
- `getApi` / `postApi` signatures are identical to legacy (`endpoint`, `params?`, `logged?`)
- `DEFAULT_URL` and `URL_KEY` match legacy values exactly
- `setBaseUrl()` added for the Settings screen (phase 9)
