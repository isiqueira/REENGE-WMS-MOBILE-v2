```typescript
import { Injectable } from '@angular/core';
import { LoadingController, ToastController, AlertController } from '@ionic/angular';

@Injectable({ providedIn: 'root' })
export class BasePageService {
  private readonly TOAST_DURATION = 2500;
  private audioError: HTMLAudioElement;
  private audioSuccess: HTMLAudioElement;

  constructor(
    private loadingCtrl: LoadingController,
    private toastCtrl: ToastController,
    private alertCtrl: AlertController,
  ) {
    this.audioError = new Audio('assets/sounds/mp3/system-fault.mp3');
    this.audioError.volume = 1.0;
    this.audioSuccess = new Audio('assets/sounds/mp3/success-5.mp3');
    this.audioSuccess.volume = 1.0;
  }

  async newLoading(): Promise<HTMLIonLoadingElement> {
    return this.loadingCtrl.create({ message: 'Por favor, aguarde ...' });
  }

  async presentToast(message: string): Promise<void> {
    const toast = await this.toastCtrl.create({
      message,
      duration: this.TOAST_DURATION,
      position: 'bottom',
    });
    await toast.present();
  }

  async presentAlert(title: string, message: string, handler?: () => void): Promise<void> {
    const alert = await this.alertCtrl.create({
      header: title,
      message,
      buttons: [{ text: 'OK', handler }],
    });
    await alert.present();
  }

  playSuccess(): void {
    this.audioSuccess.currentTime = 0;
    this.audioSuccess.play().catch(() => { /* autoplay policy — silent fail */ });
  }

  playError(): void {
    this.audioError.currentTime = 0;
    this.audioError.play().catch(() => { /* autoplay policy — silent fail */ });
  }
}
```

Notes:
- `providedIn: 'root'` — no need to declare in CoreModule providers
- Audio src paths match legacy exactly: `assets/sounds/mp3/system-fault.mp3` and `assets/sounds/mp3/success-5.mp3`
- `currentTime = 0` before play allows replaying the same sound rapidly (barcode scans in quick succession)
- Play errors silently caught (browser autoplay policy may block first play before user interaction)
